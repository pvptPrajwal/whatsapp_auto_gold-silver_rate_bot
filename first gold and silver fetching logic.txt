import time
import re
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager

def test_silver_only():
    print("🚀 Chrome Start ho raha hai Silver Rate ke liye...")
    service = Service(ChromeDriverManager().install())
    options = webdriver.ChromeOptions()
    driver = webdriver.Chrome(service=service, options=options)
    
    try:
        print("🌐 shreenavratnabullions.com khol raha hu...")
        driver.get("https://shreenavratnabullions.com/liverates.html")
        
        print("⏳ 8 Second wait kar raha hu...")
        time.sleep(8) 
        
        # --- SILVER TAB PAR CLICK ---
        print("🖱️ 'SILVER' tab dhoondh kar click kar raha hu...")
        try:
            silver_tabs = driver.find_elements(By.XPATH, "//*[text()='SILVER']")
            for tab in silver_tabs:
                if tab.is_displayed():
                    tab.click()
                    print("✅ SILVER tab par click ho gaya!")
                    time.sleep(3) 
                    break
        except Exception as e:
            pass

        # Table dhoondho
        table = driver.find_element(By.ID, "gvData_Trending_Silverr")
        rows = table.find_elements(By.TAG_NAME, "tr")
        found = False
        
        print("\n🔍 Exact 3rd Dabba (SELL Column) scan kar raha hu...\n")
        
        for row in rows:
            # Row ke saare dabbe (columns) nikal lo
            columns = row.find_elements(By.TAG_NAME, "td")
            
            # Agar row me kam se kam 3 dabbe hain
            if len(columns) >= 3:
                commodity_name = columns[0].text.upper()
                
                # Sirf us line me ghuso jisme SILVER likha ho
                if "SILVER" in commodity_name:
                    print(f"👉 Row Mili: {commodity_name.replace(chr(10), ' ')}")
                    
                    # EXACT 3rd DABBA (Index 2) = SELL Column
                    sell_box_text = columns[2].text
                    print(f"📦 SELL Box ke andar bot ko ye dikha: {sell_box_text.replace(chr(10), ' | ')}")
                    
                    # Comma hata do (taaki 2,54,148 -> 254148 ban jaye)
                    clean_text = sell_box_text.replace(',', '')
                    
                    # Us dabbe me se saare numbers nikal lo (Main Rate, High, Low sab aayenge)
                    numbers = re.findall(r'\b\d{5,7}\b', clean_text)
                    
                    if numbers:
                        # numbers[0] ka matlab hai us dabbe ka sabse PEHLA number
                        # (Aur photo me sabse upar/pehla number wahi hai jo tujhe chahiye)
                        silver_rate = int(numbers[0])
                        
                        print("="*50)
                        print(f"🎯 100% PERFECT RATE MILA: ₹ {silver_rate}")
                        print("="*50)
                        
                        found = True
                        break # Pehla kaam ka rate milte hi loop band kar do
                        
        if not found:
            print("❌ Koi rate nahi mila!")
            
    except Exception as e:
        print(f"❌ ERROR AAYA: {e}")
    finally:
        driver.quit()
        print("\n🛑 Browser band ho gaya.")

if __name__ == "__main__":
    test_silver_only()
